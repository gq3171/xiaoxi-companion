cmake_minimum_required(VERSION 3.20)
project(deskpet_scene_preview C CXX)
set(LV_BUILD_CONF_PATH ${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../../managed_components/lvgl__lvgl lvgl)
set(MAIN ${CMAKE_CURRENT_SOURCE_DIR}/../../main)
set(PANEL_SOURCES ${MAIN}/character/pet_character_assets.c ${MAIN}/fonts/pet_font_cn_16.c)
if(DESKPET_TEST_REMOTE_PANEL)
    set(PANEL_SOURCES ${MAIN}/panel/pet_panel_art.c ${MAIN}/panel/pet_panel_font.c ${MAIN}/pet_panel.c ${MAIN}/pet_panel_cache.c)
endif()
add_executable(preview preview.c ${MAIN}/pet_tts.c ${MAIN}/pet_screen.c ${MAIN}/pet_orientation.c ${MAIN}/pet_scene.c ${MAIN}/pet_home.c ${MAIN}/pet_wake_gate.c ${MAIN}/pet_asset_codec.c ${MAIN}/pet_util.c ${MAIN}/pet_turn.c ${PANEL_SOURCES} ${MAIN}/fonts/pet_font_owner_28.c ${MAIN}/fonts/pet_font_symbols_16.c ${MAIN}/../managed_components/espressif__cjson/cJSON/cJSON.c)
target_include_directories(preview PRIVATE include ${MAIN} ${MAIN}/../managed_components/espressif__cjson/cJSON)
target_compile_definitions(preview PRIVATE LV_LVGL_H_INCLUDE_SIMPLE)
target_link_libraries(preview PRIVATE lvgl m z)

if(DESKPET_TEST_REMOTE_PANEL)
    target_compile_definitions(preview PRIVATE CONFIG_DESKPET_PANEL_CACHE=1)
endif()
